{{/* @helper: Format Date with i18n support @description: Returns a date formatted in the page language. Uses the i18n "date_format" key if available; otherwise "January 2, 2006" and "Jan 2, 2006" become the locale formats :date_long and :date_medium (native month names and order), any other format is used as given with localized month names. @params: .date (time.Time): The date to format .format (string): Optional fallback Go date format (default: "Jan 2, 2006") @returns: Formatted date string @example: {{ partial "helpers/format-date.html" (dict "date" .Date) }} {{ partial "helpers/format-date.html" (dict "date" .Date "format" "January 2, 2006") }} */}} {{- $date := .date -}} {{- $fallback := .format | default "Jan 2, 2006" -}} {{- $localeFormats := dict "January 2, 2006" ":date_long" "Jan 2, 2006" ":date_medium" -}} {{- $format := i18n "date_format" | default (index $localeFormats $fallback) | default $fallback -}} {{- time.Format $format $date -}}
