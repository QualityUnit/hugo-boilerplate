{{/* Emit only actual language equivalents. Hugo pages are authoritative for languages in this build; the translation URL data covers languages built separately (for example, on another domain). Never substitute a homepage for a missing translation. */}} {{- $page := .currentPage -}} {{- if not (or $page.Params.private (eq $page.Kind "404")) -}} {{- $urls := dict $page.Language.Lang $page.Permalink -}} {{- range $page.Translations -}} {{- $urls = merge $urls (dict .Language.Lang .Permalink) -}} {{- end -}} {{- $builtLanguages := dict -}} {{- range hugo.Sites -}} {{- $builtLanguages = merge $builtLanguages (dict .Language.Lang .Language) -}} {{- end -}} {{- $allLanguages := hugo.Data.all_languages -}} {{- with $allLanguages -}} {{- $filePath := "" -}} {{- with $page.File -}} {{- $filePath = .Path -}} {{- end -}} {{- /* Pages without a source file can still use Hugo's native translations. */ -}} {{- range $code, $language := .languages -}} {{- if and (not (index $builtLanguages $code)) $filePath -}} {{- $url := partial "helpers/get-language-url.html" (dict "code" $code "langData" $language "currentPage" $page "filePath" $filePath "requireTranslation" true ) -}} {{- with $url -}} {{- $urls = merge $urls (dict $code .) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $urls) 1 -}} {{- $links := slice -}} {{- range $code, $url := $urls -}} {{- $language := index $builtLanguages $code -}} {{- $tag := $code -}} {{- with $language -}} {{- $tag = .LanguageCode | default $code -}} {{- end -}} {{- with $allLanguages -}} {{- with index .languages $code -}} {{- $tag = .params.bcp47Lang | default $tag -}} {{- end -}} {{- end -}} {{- $links = $links | append (dict "tag" $tag "url" $url) -}} {{- end -}} {{- /* x-default is optional and must point to an actual cluster member. */ -}} {{- $defaultCode := .defaultLang -}} {{- if not $defaultCode -}} {{- with $allLanguages -}} {{- $defaultCode = .defaultLanguage -}} {{- end -}} {{- end -}} {{- with $defaultCode -}} {{- with index $urls . -}} {{- $links = $links | append (dict "tag" "x-default" "url" .) -}} {{- end -}} {{- end -}} {{- range $links }} {{- end -}} {{- end -}} {{- end -}}
